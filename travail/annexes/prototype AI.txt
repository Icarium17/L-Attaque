	value = 0

class InfoSet (observed_pieces, player_to_move):
        self.player_to_move = player_to_move
        self.pieces = tuple(
            sorted(
                (
                    p.owner,
                    tuple (p.pos[0],p.pos[1]),
                    p.rank.value //either a rank or unknown if there is no observable data
                )
                for p in observed_pieces
            )
        )
	

Class Piece(type = None, position) //chose qui reste constante dans le temps, permet de faire la determination, et garde lesquelles sont lesquelles
	self.type = type
	self.possibilities -> dictionnary of type and possibilities
	self.position = position === tuple (x, y)
	
	fun can_be_type(type) -> boolean	
		return self.possibilities[type] > 0
		
	fun random_type()
		return random type based on its possibility
		
	fun move(move:Move) -> boolean:
		self.position += movement

class ObservedPiece: //pour l'info set, juste ce qu'on sait for sure
    def __init__(self, owner, pos, rank):
        self.owner = owner        # 0 = me, 1 = opponent
        self.pos = pos            # (x, y)
        self.rank = rank          # ObsRank or UNKNOWN
	
	
Class Game(belief_state:List[Pieces])
	self.types = dictionnary containing each player's dictionnary of every type with their numbers left
	self.game_state = either go through self.types and randomly pick a piece, check if it can be type and append it or for every piece in belief_state, random_type and append
	
	
	fun update(move:Move) -> None:
		apply move to game_state
		
	fun check_final() -> boolean:
		return (end_conditions are met)

	fun get_opponent_board() -> None:
		randomly generates the opponent`s pieces

			while (!final):

				if (!child[1])

				node_now = childNodes

				if (game.check_final())
